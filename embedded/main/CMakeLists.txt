# Main component for Reflex OS
#
# Build targets:
#   - gie (default): GIE + LP core + VDB + feedback (Board A / receiver)
#   - sender:        ESP-NOW pattern sender (Board B)
#
# Build sender:  idf.py -DREFLEX_TARGET=sender build
# Build GIE:     idf.py build   (or -DREFLEX_TARGET=gie)

if(NOT DEFINED REFLEX_TARGET)
    set(REFLEX_TARGET "gie")
endif()

if(REFLEX_TARGET STREQUAL "sender")
    set(app_sources "espnow_sender.c")
    # Pass TRANSITION_MODE to C preprocessor if set via -DTRANSITION_MODE=1
    if(DEFINED TRANSITION_MODE AND TRANSITION_MODE)
        add_compile_definitions(TRANSITION_MODE=1)
    endif()
else()
    set(app_sources
        "gie_engine.c"
        "geometry_cfc_freerun.c"
        "test_gie_core.c"
        "test_espnow.c"
        "test_live_input.c"
        "test_memory.c"
        "test_kinetic.c"
        "test_lp_char.c"
        "test_hebbian.c"
        "reflex_vdb.c"
        "reflex_espnow.c"
    )
    # Pass LP_SEED to C preprocessor if set via -DLP_SEED=0xDEAD5678
    if(DEFINED LP_SEED)
        add_compile_definitions(LP_SEED=${LP_SEED})
    endif()
    # Skip Tests 1-13, run only Test 11 (enrollment) + Test 14C
    if(DEFINED SKIP_TO_14C AND SKIP_TO_14C)
        add_compile_definitions(SKIP_TO_14C=1)
    endif()
    # R3a: mask pattern_id trits [16..23] from signatures
    if(DEFINED MASK_PATTERN_ID AND MASK_PATTERN_ID)
        add_compile_definitions(MASK_PATTERN_ID=1)
    endif()
    # R3a follow-up: mask RSSI thermometer [0..15] from signatures
    if(DEFINED MASK_RSSI AND MASK_RSSI)
        add_compile_definitions(MASK_RSSI=1)
    endif()
    # H2: mask pattern_id trits [16..23] from the INPUT (not just signatures)
    # Makes GIE hidden state genuinely label-free for Hebbian leakage test
    if(DEFINED MASK_PATTERN_ID_INPUT AND MASK_PATTERN_ID_INPUT)
        add_compile_definitions(MASK_PATTERN_ID_INPUT=1)
    endif()
    # Audit (Sep 2026): mask sequence trits [104..119] from the INPUT in
    # espnow_encode_rx_entry(). The legacy encoder espnow_encode_input()
    # already zeroes them; the stream encoder used by every load-bearing test
    # does not. Sequence is monotonic while patterns cycle, so it is a
    # possible back-channel into the GIE hidden state -> VDB -> LP path on
    # which the divergence measurements are taken. Default OFF so the
    # firmware continues to match the committed apr9/apr11 datasets; turn ON
    # to run the ablation and confirm the MTFP baseline is unchanged.
    if(DEFINED MASK_SEQUENCE_INPUT AND MASK_SEQUENCE_INPUT)
        add_compile_definitions(MASK_SEQUENCE_INPUT=1)
    endif()
    # Skip to Test 14 (enrollment + kinetic attention only)
    if(DEFINED SKIP_TO_14 AND SKIP_TO_14)
        add_compile_definitions(SKIP_TO_14=1)
    endif()
    # Skip to Test 15 (enrollment + Hebbian only)
    if(DEFINED SKIP_TO_15 AND SKIP_TO_15)
        add_compile_definitions(SKIP_TO_15=1)
    endif()
    # TriX integration: dump each (pattern_id, 128-trit input) during the
    # 30s Phase 0a observation as "TRAIN p=<id> v=<128 chars +/-/0>" lines
    # over UART. Used by tools/reflex_signature_train.py in trixV to build
    # a training dataset for QAT comparison against sign-of-mean.
    if(DEFINED TRIX_DUMP_TRAINING AND TRIX_DUMP_TRAINING)
        add_compile_definitions(TRIX_DUMP_TRAINING=1)
    endif()
    # TriX integration: override sign-of-mean signatures with externally
    # trained TriX signatures from main/trix_signatures.h (generated by
    # trixV's tools/reflex_signature_train.py). The receiver still runs
    # Phase 0a-0b for diagnostic stats but the on-silicon classifier
    # uses the override.
    if(DEFINED USE_TRIX_WEIGHTS AND USE_TRIX_WEIGHTS)
        add_compile_definitions(USE_TRIX_WEIGHTS=1)
    endif()
    # TriX integration: disable the per-packet novelty gate in Phase 1.
    # The fixed NOVELTY_THRESHOLD=60 is calibrated for sign-of-mean
    # signature density; QAT-trained signatures may be sparser and
    # produce smaller absolute scores. For the falsification protocol
    # comparison, classification accuracy is the variable of interest —
    # novelty is an orthogonal mechanism. Both baseline and TriX runs
    # should use the same flag to be apples-to-apples.
    if(DEFINED TRIX_DISABLE_NOVELTY AND TRIX_DISABLE_NOVELTY)
        add_compile_definitions(TRIX_DISABLE_NOVELTY=1)
    endif()
endif()

idf_component_register(
    SRCS
        ${app_sources}
    INCLUDE_DIRS
        "../include"
        "../shared"
    REQUIRES
        driver
        esp_timer
        esp_driver_gptimer
        esp_driver_rmt
        esp_driver_pcnt
        esp_driver_parlio
        esp_driver_ledc
        esp_hw_support
        esp_wifi
        esp_event
        esp_netif
        nvs_flash
        ulp
    WHOLE_ARCHIVE
)

# LP Core geometric processor (GIE target only)
if(NOT REFLEX_TARGET STREQUAL "sender")
    set(ulp_app_name ulp_${COMPONENT_NAME})
    set(ulp_sources "ulp/main.S")
    set(ulp_exp_dep_srcs ${app_sources})
    ulp_embed_binary(${ulp_app_name} "${ulp_sources}" "${ulp_exp_dep_srcs}")
endif()
